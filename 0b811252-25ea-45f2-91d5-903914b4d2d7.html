            const [gameState, setGameState] = useState('welcome'); // welcome, playing, finished
            const [currentStationIdx, setCurrentStationIdx] = useState(0);
            const [timePerStation, setTimePerStation] = useState(60); // default 60 sec
            const [timeLeft, setTimeLeft] = useState(60);
            const [answers, setAnswers] = useState({});
            const [isAutoRotate, setIsAutoRotate] = useState(true);
            const [studentName, setStudentName] = useState('');
            const [studentClass, setStudentClass] = useState('');

            const timerRef = useRef(null);

            // Initialize audio for bell/knock effect
            const playSound = () => {
                try {
                    const ctx = new (window.AudioContext || window.webkitAudioContext)();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(800, ctx.currentTime);
                    gain.gain.setValueAtTime(0.3, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.5);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.5);
                } catch(e) {
                    console.log('Audio error:', e);
                }
            };

            // Timer Tick
            useEffect(() => {
                if (gameState === 'playing' && isAutoRotate) {
                    timerRef.current = setInterval(() => {
                        setTimeLeft((prev) => {
                            if (prev <= 1) {
                                playSound();
                                handleNextStation();
                                return timePerStation;
                            }
                            return prev - 1;
                        });
                    }, 1000);
                }
                return () => clearInterval(timerRef.current);
            }, [gameState, currentStationIdx, isAutoRotate, timePerStation]);

            const handleStartQuiz = () => {
                if (!studentName.trim()) {
                    alert('Harap masukkan nama peserta terlebih dahulu!');
                    return;
                }
                setGameState('playing');
                setCurrentStationIdx(0);
                setTimeLeft(timePerStation);
            };

            const handleInputChange = (blankId, val) => {
                setAnswers(prev => ({
                    ...prev,
                    [blankId]: val
                }));
            };

            const handleNextStation = () => {
                if (currentStationIdx < quizData.length - 1) {
                    setCurrentStationIdx(prev => prev + 1);
                    setTimeLeft(timePerStation);
                } else {
                    setGameState('finished');
                }
            };

            const handlePrevStation = () => {
                if (currentStationIdx > 0) {
                    setCurrentStationIdx(prev => prev - 1);
                    setTimeLeft(timePerStation);
                }
            };

            const calculateScore = () => {
                let totalCorrect = 0;
                let totalQuestions = 75; // 25 stations x 3 blanks

                quizData.forEach(st => {
                    st.blanks.forEach(b => {
                        const userAns = (answers[b.id] || '').trim().toLowerCase();
                        const isMatch = b.keys.some(k => k.toLowerCase() === userAns);
                        if (isMatch) totalCorrect++;
                    });
                });

                const finalScore = Math.round((totalCorrect / totalQuestions) * 100);
                return { totalCorrect, totalQuestions, finalScore };
            };

            // Render Views
            if (gameState === 'welcome') {
                return (
                    <div className="min-h-screen flex items-center justify-center p-4">

                            <div className="space-y-4 text-left mb-6 bg-slate-50 p-4 rounded-xl text-sm text-slate-600 border border-slate-200">
                                <p className="font-semibold text-slate-700">Instruksi Praktikum:</p>
                                <ul className="list-disc list-inside space-y-1">
                                    <li>Terdapat <b>25 Stasiun (Pos)</b> soal berpindah otomatis.</li>
                                    <li>Setiap stasiun memiliki <b>3 isian singkat</b>.</li>
                                    <li>Ketukan bel/detik akan berbunyi saat waktu stasiun habis.</li>

                                        placeholder="Masukkan nama lengkap"

                                    />
